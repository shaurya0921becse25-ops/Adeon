    function handleFormSubmit(e) {
      e.preventDefault();

      const title = document.getElementById('form-title').value.trim();
      const code = document.getElementById('form-code').value.trim();
      const notes = document.getElementById('form-notes').value.trim();
      const status = document.getElementById('form-status').value;
      const from = document.getElementById('form-from').value.trim();
      const to = document.getElementById('form-to').value.trim();
      const carrier = document.getElementById('form-carrier').value.trim();
      const flightNo = document.getElementById('form-flight-no').value.trim();
      const seat = document.getElementById('form-seat').value.trim();

      const cat = selectedBookingType.toLowerCase() === 'flight' ? 'flight' 
                : selectedBookingType.toLowerCase() === 'car' ? 'car' 
                : 'all';

      if (currentEditId) {
        // Update
        const idx = transports.findIndex(t => t.id === currentEditId);
        if (idx !== -1) {
          transports[idx] = {
            ...transports[idx],
            type: selectedBookingType,
            title,
            code,
            notes,
            status,
            category: cat,
            from,
            to,
            carrier,
            flightNo,
            seat
          };
        }
      } else {
        // Create
        const newItem = {
          id: Date.now().toString(),
          type: selectedBookingType,
          status: status || 'Confirmed',
          title: title || `${selectedBookingType} Booking`,
          code,
          notes,
          category: cat,
          from,
          to,
          carrier,
          flightNo,
          seat
        };
        transports.unshift(newItem);
      }

      updateCounts();
      renderCards();
      closeModal();
    }
