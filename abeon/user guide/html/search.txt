    <script>
        // Initialize Lucide Icons
        lucide.createIcons();

        // Dropdown Toggle Handlers
        const whereToField = document.getElementById('whereToField');
        const whenField = document.getElementById('whenField');
        const recentSearchesModal = document.getElementById('recentSearchesModal');
        const datepickerModal = document.getElementById('datepickerModal');
        const whereInput = document.getElementById('whereInput');
        const heroSection = document.querySelector('.hero-section');
        const updateDropdownState = () => {
            heroSection.classList.toggle(
                'dropdown-open',
                recentSearchesModal.classList.contains('active') || datepickerModal.classList.contains('active')
            );
        };

        whereToField.addEventListener('click', (e) => {
            e.stopPropagation();
            datepickerModal.classList.remove('active');
            recentSearchesModal.classList.toggle('active');
            updateDropdownState();
        });

        whenField.addEventListener('click', (e) => {
            e.stopPropagation();
            recentSearchesModal.classList.remove('active');
            datepickerModal.classList.toggle('active');
            updateDropdownState();
        });

        // Close modals when clicking outside
        document.addEventListener('click', () => {
            recentSearchesModal.classList.remove('active');
            datepickerModal.classList.remove('active');
            updateDropdownState();
        });

        recentSearchesModal.addEventListener('click', (e) => {
            e.stopPropagation();
        });

        datepickerModal.addEventListener('click', (e) => {
            e.stopPropagation();
        });

        // Set search field input when selecting a recent destination
        const recentItems = document.querySelectorAll('.recent-search-item');
        recentItems.forEach(item => {
            item.addEventListener('click', () => {
                const cityName = item.querySelector('strong').innerText;
                whereInput.value = cityName;
                recentSearchesModal.classList.remove('active');
                updateDropdownState();
            });
        });
    </script>
